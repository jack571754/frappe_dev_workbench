## 4. 配置阶段

### 4.1 配置来源

飞书登录配置支持两种来源，优先级如下：

**优先级 1：site_config.json（推荐）**

```json
{
  "feishu_app_id": "cli_xxxxx",
  "feishu_app_secret": "your_app_secret",
  "feishu_host": "https://open.feishu.cn"
}
```

**优先级 2：Social Login Key（回退）**

在 Frappe 后台创建 Social Login Key 记录，name 为 `feishu`。

### 4.2 自动配置流程

#### 文件：`install.py`

```python
def setup_feishu_social_login():
    """
    设置飞书社交登录配置
    从 site_config.json 读取 feishu_app_id 和 feishu_app_secret
    如有配置则自动启用飞书登录
    """
    from product_sales_planning.api.v1.feishu_setup import get_feishu_provider_config

    provider_name = "feishu"

    app_id = frappe.conf.get("feishu_app_id", "")
    app_secret = frappe.conf.get("feishu_app_secret", "")

    if not app_id or not app_secret:
        return

    if frappe.db.exists("Social Login Key", provider_name):
        doc = frappe.get_doc("Social Login Key", provider_name)
        if doc.client_id != app_id or not doc.enable_social_login:
            doc.client_id = app_id
            doc.client_secret = app_secret
            doc.enable_social_login = 1
            doc.save(ignore_permissions=True)
            print("✅ 飞书登录配置已更新并启用")
        return

    config = get_feishu_provider_config()
    doc = frappe.new_doc("Social Login Key")
    doc.update(config)
    doc.name = provider_name
    doc.client_id = app_id
    doc.client_secret = app_secret
    doc.enable_social_login = 1
    doc.insert(ignore_permissions=True)
    print("✅ 飞书登录配置已创建并启用")
```

**触发时机**：
- `after_install()` - 应用安装后
- `after_migrate()` - 数据库迁移后

## 5. 授权入口阶段

### 5.1 登录页面上下文修改

#### 文件：`overrides/oauth.py`

```python
def update_login_context(context):
    """
    更新登录页面的 context，修改飞书的授权 URL

    这个函数通过 update_website_context hook 被调用

    为什么需要覆盖？
    - 飞书的 OAuth 授权 URL 需要使用 app_id 参数
    - Frappe 默认使用 client_id 参数
    - 因此需要自定义授权 URL
    """
    # 只处理登录页面
    if not context.get("provider_logins"):
        return

    # 获取当前请求的 redirect-to 参数
    redirect_to = frappe.local.request.args.get("redirect-to") if frappe.local.request else None

    # 修改飞书的 auth_url
    for provider in context.get("provider_logins", []):
        if provider.get("name") == "feishu":
            provider["auth_url"] = get_feishu_authorize_url(redirect_to)
```

### 5.2 生成飞书授权 URL

```python
def get_feishu_authorize_url(redirect_to: str = None) -> str:
    """
    获取飞书授权 URL

    直接调用自定义的 feishu_login.authorize 端点

    Args:
        redirect_to: 登录后跳转的目标路径

    Returns:
        飞书授权入口 URL
    """
    base_url = frappe.utils.get_url()

    # 构建指向自定义 authorize 端点的 URL
    params = {}
    if redirect_to:
        # 确保 redirect_to 是相对路径，避免端口丢失问题
        if redirect_to.startswith("http"):
            from urllib.parse import urlparse
            parsed = urlparse(redirect_to)
            redirect_to = parsed.path or "/planning/"
        params["redirect_to"] = redirect_to

    authorize_url = f"{base_url}/api/method/product_sales_planning.api.v1.feishu_login.authorize"

    if params:
        authorize_url = f"{authorize_url}?{urlencode(params)}"

    return authorize_url
```

### 5.3 authorize 接口详解

#### 文件：`api/v1/feishu_login.py`

**接口路径**：`/api/method/product_sales_planning.api.v1.feishu_login.authorize`

**访问方式**：GET（允许游客访问）

```python
@frappe.whitelist(allow_guest=True)
def authorize(redirect_to: str = None):
    """
    飞书授权入口 - 跳转到飞书授权页面

    访问: /api/method/product_sales_planning.api.v1.feishu_login.authorize?redirect_to=/planning/

    执行流程：
    1. 获取飞书配置（App ID, App Secret）
    2. 构建 state 参数（包含站点信息、token、跳转路径）
    3. 生成飞书授权 URL
    4. 返回 302 重定向
    """
    try:
        # 1. 获取飞书配置
        settings = get_feishu_settings()

        # 2. 处理 redirect_to 参数
        if redirect_to and redirect_to.startswith("http"):
            from urllib.parse import urlparse
            parsed = urlparse(redirect_to)
            redirect_to = parsed.path or "/planning/"

        # 3. 构建 state 参数（防止 CSRF 攻击）
        state = {
            "site": frappe.utils.get_url(),
            "token": frappe.generate_hash(),
            "redirect_to": redirect_to or "/planning/",
        }

        # 4. 构建飞书授权 URL 参数
        params = {
            "app_id": settings["app_id"],
            "redirect_uri": _get_redirect_uri(),
            "state": base64.b64encode(json.dumps(state).encode("utf-8")).decode("utf-8"),
        }

        # 5. 飞书网页授权 URL
        authorize_url = f"{settings['feishu_host']}/open-apis/authen/v1/authorize?{urlencode(params)}"

        # 6. 返回 302 重定向
        frappe.local.response["type"] = "redirect"
        frappe.local.response["location"] = authorize_url

    except Exception as e:
        frappe.log_error(title="Feishu Authorize Error", message=frappe.get_traceback())
        frappe.respond_as_web_page(
            _("飞书登录失败"),
            _("无法获取飞书授权链接: {0}").format(str(e)),
            http_status_code=500
        )
```

### 5.4 获取配置方法

```python
def get_feishu_settings():
    """
    获取飞书登录配置

    优先从 site_config.json 读取，如果没有则从 Social Login Key 读取

    Returns:
        dict: {
            "app_id": "cli_xxxxx",
            "app_secret": "your_secret",
            "feishu_host": "https://open.feishu.cn"
        }
    """
    # 优先从 site_config.json 读取
    app_id = frappe.conf.get("feishu_app_id")
    app_secret = frappe.conf.get("feishu_app_secret")

    if app_id and app_secret:
        return {
            "app_id": app_id,
            "app_secret": app_secret,
            "feishu_host": frappe.conf.get("feishu_host", FEISHU_HOST),
        }

    # 回退到 Social Login Key
    provider = "feishu"
    if not frappe.db.exists("Social Login Key", provider):
        frappe.throw(_("飞书登录未配置，请在 site_config.json 中添加 feishu_app_id 和 feishu_app_secret"))

    from frappe.utils.password import get_decrypted_password

    doc = frappe.get_doc("Social Login Key", provider)

    return {
        "app_id": doc.client_id,
        "app_secret": get_decrypted_password("Social Login Key", provider, "client_secret"),
        "feishu_host": doc.base_url or FEISHU_HOST,
    }
```

## 6. 飞书授权页面

### 6.1 飞书授权 URL 格式

```
https://open.feishu.cn/open-apis/authen/v1/authorize?app_id=cli_xxxxx&redirect_uri=https://your-site.com/api/method/product_sales_planning.api.v1.feishu_login.callback&state=base64_encoded_state
```

### 6.2 参数说明

| 参数 | 说明 | 来源 |
|------|------|------|
| `app_id` | 飞书应用 ID | site_config.json 或 Social Login Key |
| `redirect_uri` | 回调地址 | 固定为 `/api/method/.../callback` |
| `state` | 状态参数 | Base64 编码的 JSON 对象 |

### 6.3 state 参数结构

```json
{
  "site": "https://your-site.com",
  "token": "random_hash_for_csrf_protection",
  "redirect_to": "/planning/"
}
```

### 6.4 用户授权流程

用户在飞书授权页面可以：
- **同意授权**：飞书重定向到 callback 接口，携带 `code` 和 `state`
- **拒绝授权**：飞书重定向到 callback 接口，携带 `error` 和 `error_description`

## 7. 回调处理阶段

### 7.1 callback 接口详解

#### 文件：`api/v1/feishu_login.py`

**接口路径**：`/api/method/product_sales_planning.api.v1.feishu_login.callback`

**访问方式**：GET（允许游客访问）

```python
@frappe.whitelist(allow_guest=True)
def callback(code: str = None, state: str = None, error: str = None, error_description: str = None):
    """
    飞书授权回调 - 处理飞书返回的授权码

    飞书会回调: /api/method/.../callback?code=xxx&state=xxx
    如果用户拒绝授权，飞书会返回 error 和 error_description 参数

    执行流程：
    1. 检查是否有错误（用户拒绝授权）
    2. 验证授权码 code
    3. 用 code 换取 user_access_token
    4. 用 user_access_token 获取用户信息
    5. 创建或更新用户
    6. 执行登录
    7. 重定向到目标页面
    """
```

### 7.3 获取 Token 流程

```python
def _get_user_access_token(settings, code):
    """
    用授权码换取 user_access_token

    步骤：
    1. 先获取 app_access_token（应用级别的 token）
    2. 用 app_access_token + code 换取 user_access_token

    Args:
        settings: 飞书配置
        code: 授权码

    Returns:
        user_access_token
    """
    # 步骤 1: 获取 app_access_token
    app_access_token = _get_app_access_token(settings)

    # 步骤 2: 用 code 换取 user_access_token
    url = f"{settings['feishu_host']}{USER_ACCESS_TOKEN_URI}"
    headers = {
        "Content-Type": "application/json",
        "Authorization": f"Bearer {app_access_token}",
    }
    req_body = {
        "grant_type": "authorization_code",
        "code": code
    }

    response = requests.post(url=url, headers=headers, json=req_body, timeout=10)
    _check_response(response)

    return response.json().get("data", {}).get("access_token")
```

```python
def _normalize_mobile(mobile: str) -> str:
    """
    标准化手机号，移除国际区号前缀

    Args:
        mobile: 原始手机号，可能带有 +86 等前缀

    Returns:
        标准化后的手机号（11位）
    """
    if not mobile:
        return ""
    # 移除空格和横线
    mobile = mobile.replace(" ", "").replace("-", "")
    # 移除 + 号和国际区号
    if mobile.startswith("+86"):
        mobile = mobile[3:]
    elif mobile.startswith("+"):
        mobile = mobile[1:]
    if mobile.startswith("86") and len(mobile) > 11:
        mobile = mobile[2:]
    return mobile
```

## 10. 前端路由守卫

### 10.1 权限守卫创建

#### 文件：`frontend/src/router/guards/permissionGuard.js`

```javascript
/**
 * Permission Route Guard
 * 路由守卫，控制页面访问权限
 */
export function createPermissionGuard(router, permissionStore) {
  router.beforeEach(async (to, _from, next) => {
    try {
      // 1. 确保权限已加载
      if (!permissionStore.isLoaded) {
        try {
          await permissionStore.loadPermissions()
        } catch (error) {
          console.error('加载权限失败:', error)
          // 权限加载失败，重定向到登录页
          window.location.href = '/login?redirect=' + encodeURIComponent(to.fullPath)
          return
        }
      }

      // 2. 获取路由配置
      const pageConfig = getPageConfig(to.name)

      // 3. 检查是否需要登录
      if (pageConfig?.requireAuth && !permissionStore.user) {
        window.location.href = '/login?redirect=' + encodeURIComponent(to.fullPath)
        return
      }

      // 4. 检查路由权限
      const hasAccess = canAccessRoute(to.name, permissionStore.roles)

      if (!hasAccess) {
        next({ name: 'NoAccess', query: { from: to.name } })
        return
      }

      // 5. 允许访问
      next()
    } catch (error) {
      console.error('Permission guard error:', error)
      next()
    }
  })
}
```

### 10.2 权限检查逻辑

```javascript
/**
 * 检查用户是否可以访问路由
 */
export function canAccessRoute(routeName, userRoles) {
  const config = getPageConfig(routeName)

  if (!config) return true
  if (!config.requireAuth) return true
  if (config.roles.includes('*')) return true  // 所有登录用户可访问

  if (!Array.isArray(userRoles)) {
    return false
  }

  // 用户具有任一所需角色即可访问
  return config.roles.some((role) => userRoles.includes(role))
}
```

## 16. 常见问题

### Q1: 飞书登录失败，提示 "未配置"

**原因**：site_config.json 中没有配置 `feishu_app_id` 和 `feishu_app_secret`

**解决**：在 site_config.json 中添加配置，或通过 Frappe Console 调用 `setup_feishu_login()`

### Q2: 用户创建后没有角色

**原因**：新用户默认添加 `Planning Admin` 角色

**解决**：检查角色是否存在，或手动为用户分配角色

### Q3: 飞书通知发送失败

**可能原因**：
1. 用户未通过飞书登录（没有 open_id 关联）
2. 用户手机号与飞书不匹配

**解决**：
1. 让用户重新使用飞书登录
2. 检查用户的 social_logins 中是否有 feishu 记录

### Q4: 登录后重定向到错误的页面

**原因**：`redirect_to` 参数处理问题

**解决**：确保 `redirect_to` 是相对路径，检查 `host_name` 配置
